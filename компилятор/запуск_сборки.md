# Запуск сборщика

```bash
set -Eeuo pipefail

echo "== BUILDER RUN START =="
date

python3 компилятор/import_codesys_FINAL.py > компилятор/logs/final_test.log 2>&1

echo "== FINAL LOG TAIL =="
tail -n 200 компилятор/logs/final_test.log || true

echo "== GREP ERRORS =="
grep -En "error|exception|traceback|warning|failed" компилятор/logs/final_test.log || true

echo "== BUILDER RUN DONE =="
date
```
