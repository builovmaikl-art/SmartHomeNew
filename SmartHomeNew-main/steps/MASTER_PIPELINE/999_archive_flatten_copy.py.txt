from pathlib import Path
import shutil
import sys

ROOT = Path.cwd()
ARCHIVE = ROOT / 'archive'
REPORT = ROOT / 'workspace' / 'logs' / '2026-04-11_archive_flatten_report.txt'


def pick_name(name: str) -> str:
    target = ROOT / name
    if not target.exists():
        return name
    p = Path(name)
    suffix = ''.join(p.suffixes)
    stem = name[:-len(suffix)] if suffix else name
    i = 1
    while True:
        candidate = f'{stem}__archive_{i}{suffix}'
        if not (ROOT / candidate).exists():
            return candidate
        i += 1


def main() -> int:
    if not ARCHIVE.exists() or not ARCHIVE.is_dir():
        print("ERROR: archive directory not found")
        return 2

    REPORT.parent.mkdir(parents=True, exist_ok=True)
    files = sorted(p for p in ARCHIVE.rglob('*') if p.is_file())
    report = [f'SOURCE_FILE_COUNT={len(files)}']

    for src in files:
        final_name = pick_name(src.name)
        dst = ROOT / final_name
        shutil.copy2(src, dst)
        state = 'RENAMED' if final_name != src.name else 'COPIED'
        line = f'{state}\t{src.relative_to(ROOT)}\t{final_name}'
        print(line)
        report.append(line)

    REPORT.write_text('\n'.join(report) + '\n', encoding='utf-8')
    print(f'REPORT={REPORT}')
    print('OK')
    return 0


if __name__ == '__main__':
    sys.exit(main())
